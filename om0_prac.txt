{$CLEO .cs}
{$USE ini}
nop

function CControllerConfigManager_GetIsKeyboardKeyJustDown<thiscall, 0x52E450>(self: int {CControllerConfigManager}, key: int {RsKeyCodes}): int

int timer = 0
int offset = 29

const CONFIG_PATH = "cleo\\om0_prac.ini"

if not does_file_exist CONFIG_PATH
then
    File config_file = open_file CONFIG_PATH "at"
    write_string_to_file config_file "; List of valid codes: https://learn.microsoft.com/en-us/windows/win32/inputdev/virtual-key-codes"
    close_file config_file
    write_int_to_ini_file 70 CONFIG_PATH "Keys" "EnterExit"
    write_int_to_ini_file 50 CONFIG_PATH "Keys" "Submission"
end

int enterExitKey = read_int_from_ini_file CONFIG_PATH "Keys" "EnterExit"
int submissionKey = read_int_from_ini_file CONFIG_PATH "Keys" "Submission"

while true
    wait 0
    int waitAddress = 0xA75930
    write_memory waitAddress 1 0 0

    if
        IS_CHAR_IN_ANY_CAR $scplayer
    then
        Car currentCar
        GET_CAR_CHAR_IS_USING $scplayer currentCar

        int modelId
        get_car_model currentCar modelId

        // 523: Cop Bike Model ID
        if
            modelId <> 523
        then
            offset = 49
        else
            offset = 29
        end
    end
    
    // Check if player is using a mouse button. then add 1 to offset to accomodate for the lack of input delay
    if or
        submissionKey == 1
        submissionKey == 2
        submissionKey == 4
        submissionKey == 5
        submissionKey == 6
    then
        offset += 1
    end

    int isEnterExitKeyJustPressed = CControllerConfigManager_GetIsKeyboardKeyJustDown(0xB70198, enterExitKey)
    if and
        is_button_pressed PadId.Pad1 Button.Rightshock
        isEnterExitKeyJustPressed == 12006401
    then
        while is_button_pressed PadId.Pad1 Button.Rightshock
            wait 0
            timer++
        end
        wait 400
        clear_prints
        int difference = timer - offset
        //        trace "%d" difference
        if
            difference == 0
        then
            print_string_now "YAY" 2000
        else
            if
                difference < 0
            then
                difference = difference * -1
                print_formatted "You were %d frame(s) too early" 2000 difference
            else
                print_formatted "You were %d frame(s) too late" 2000 difference
            end
        end
        timer = 0
    end
end

terminate_this_custom_script
